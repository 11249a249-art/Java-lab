class Boarder {
    String name;
    int days;

   
    Boarder(String name, int days) {
        this.name = name;
        this.days = days;
    }

   
    double bill() {
        double amount = days * 85.0;

        if (days < 26) {
            amount = amount * 0.90;   
        }

        return amount;
    }

    
    void display() {
        System.out.printf("%-10s %4d %8.2f%n",
                name, days, bill());
    }
}

public class tryex1 {
    public static void main(String[] args) {

        
        Boarder b1 = new Boarder("Aravind", 28);
        Boarder b2 = new Boarder("Divya", 30);
        Boarder b3 = new Boarder("Karthik", 25);

        
        System.out.println("NAME       DAYS     BILL");

        b1.display();
        b2.display();
        b3.display();

       
        Boarder highest = b1;

        if (b2.bill() > highest.bill()) {
            highest = b2;
        }

        if (b3.bill() > highest.bill()) {
            highest = b3;
        }

        
        System.out.println("\nStudent with highest bill:");
        System.out.println("Name: " + highest.name);
        System.out.println("Days: " + highest.days);
        System.out.printf("Bill: Rs. %.2f%n", highest.bill());
    }
}
